void strip_space(char *src, uint32_t line_len) {
    uint32_t p1 = 0;
    uint32_t p2 = 0;
    
    for(; p1 < line_len; ++p1) {
	    if(!isspace(src[p1])) {
		    src[p2] = src[p1];
		    ++p2;
	    }
    }
    
    src[p2] = '\0';
}
